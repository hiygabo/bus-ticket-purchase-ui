async generateTicketPDF(id_detail: number): Promise<PDFKit.PDFDocument> {
    const ticket = await this.findOneDetail(id_detail);

    const doc = new PDFDocument({ margin: 45, size: 'A4' });

    const red = '#c8102e';
    const dark = '#1a1a2e';
    const gray = '#6b6b70';
    const lightGray = '#f4f4f5';
    const borderGray = '#e0e0e3';
    const white = '#ffffff';

    // ======================================================================
    // CABECERA (banda roja con la empresa)
    // ======================================================================
    doc.rect(0, 0, doc.page.width, 95).fill(red);

    doc.fillColor(white).fontSize(24).font('Helvetica-Bold');
    doc.text('TRANS COPACABANA S.A.', 45, 22, { align: 'left' });

    doc.fontSize(10.5).font('Helvetica');
    doc.text('TRAVEL TICKET  /  BOLETO DE VIAJE', 45, 58);
    doc.text('Av. de la Amistad Nº 123 · La Paz, Bolivia', 45, 74);

    // Nº de ticket en la esquina derecha de la cabecera
    doc.fontSize(11).font('Helvetica-Bold');
    doc.text(`Nº ${ticket.id_detail}`, doc.page.width - 45, 48, { align: 'right' });

    // ======================================================================
    // SECCIÓN: PASAJERO
    // ======================================================================
    let y = 135;

    doc.fontSize(12).font('Helvetica-Bold').fillColor(dark).text('PASSENGER DATA / DATOS DEL PASAJERO', 45, y);
    y += 24;
    y = this.drawPair(doc, 'Name', ticket.passenger.full_name ?? '-', y, 45, borderGray, lightGray, dark, gray);
    y = this.drawPair(doc, 'CI', ticket.passenger.ci ?? '-', y, 45, borderGray, lightGray, dark, gray);

    // ======================================================================
    // SECCIÓN: VIAJE
    // ======================================================================
    y += 18;
    doc.fontSize(12).font('Helvetica-Bold').fillColor(dark).text('TRAVEL DETAILS / DETALLES DEL VIAJE', 45, y);
    y += 24;

    const origin = ticket.travel.travel_origin?.place?.place_name
      ?? ticket.travel.travel_origin?.stop_name
      ?? '-';
    const destiny = ticket.travel.travel_destiny?.place?.place_name
      ?? ticket.travel.travel_destiny?.stop_name
      ?? '-';

    y = this.drawPair(doc, 'Route', `${origin} → ${destiny}`, y, 45, borderGray, lightGray, dark, gray);
    y = this.drawPair(doc, 'Date', `${ticket.travel.departure_date ?? '-'}`, y, 45, borderGray, lightGray, dark, gray);
    y = this.drawPair(doc, 'Departure', `${ticket.travel.schedule?.departure_time ?? '-'}`, y, 45, borderGray, lightGray, dark, gray);
    y = this.drawPair(doc, 'Arrival', `${ticket.travel.schedule?.estimated_arrival_time ?? '-'}`, y, 45, borderGray, lightGray, dark, gray);

    // ======================================================================
    // SECCIÓN: BUS Y ASIENTO
    // ======================================================================
    y += 18;
    doc.fontSize(12).font('Helvetica-Bold').fillColor(dark).text('BUS & SEAT / BUS Y ASIENTO', 45, y);
    y += 24;

    y = this.drawPair(doc, 'Bus plate', `${ticket.travel.bus?.bus_plate ?? '-'}`, y, 45, borderGray, lightGray, dark, gray);
    y = this.drawPair(doc, 'Seat', `${ticket.seat.seat_number ?? ticket.seat.id_seat}`, y, 45, borderGray, lightGray, dark, gray);

    // ======================================================================
    // TOTAL
    // ======================================================================
    y += 28;
    const totalX = doc.page.width - 45 - 220;
    doc.rect(totalX, y, 220, 40).fill(red);
    doc.fillColor(white).font('Helvetica-Bold');
    doc.fontSize(12).text('TOTAL', totalX + 16, y + 12);
    doc.fontSize(16).text(`Bs. ${ticket.travel.price}`, totalX + 16, y + 8, { align: 'right', width: 220 - 32 });

    y += 90;

    // ======================================================================
    // PIE
    // ======================================================================
    doc.moveTo(45, y).lineTo(doc.page.width - 45, y).lineWidth(1).stroke(borderGray);
    y += 22;
    doc.fillColor(gray).font('Helvetica').fontSize(9.5);
    doc.text('Present this ticket printed or digital on your travel day.', 45, y, { align: 'center' });
    doc.text('Thank you for traveling with us!  /  ¡Gracias por viajar con nosotros!', 45, y + 16, { align: 'center' });

    return doc;
  }

  private drawPair(
    doc: PDFKit.PDFDocument,
    labelText: string,
    value: string,
    top: number,
    x: number,
    borderGray: string,
    lightGray: string,
    dark: string,
    gray: string,
  ): number {
    const h = 34;
    const labelW = 130;

    doc.save();
    doc.rect(x, top, labelW, h).fill(lightGray).stroke(borderGray);
    doc.fillColor(gray).font('Helvetica-Bold').fontSize(10);
    doc.text(labelText.toUpperCase(), x + 9, top + 10, { width: labelW - 18 });

    doc.rect(x + labelW, top, doc.page.width - x - labelW - 45, h).fillColor('#ffffff').stroke(borderGray);
    doc.fillColor(dark).font('Helvetica').fontSize(12);
    doc.text(value, x + labelW + 9, top + 9, { width: doc.page.width - x - labelW - 45 - 18 });

    doc.restore();
    return top + h;
  }
